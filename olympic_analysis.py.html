"""
Global Olympic Sport Performance Analysis
-----------------------------------------
Dataset: "120 years of Olympic history: athletes and results" (Kaggle)
Files needed in the same folder:
    athlete_events.csv
    noc_regions.csv

Install: pip install pandas matplotlib seaborn
Run:     python olympic_analysis.py
"""

import os
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns

sns.set_theme(style="whitegrid")
os.makedirs("outputs", exist_ok=True)


# ------------------------------------------------------------------
# 1. LOAD & CLEAN
# ------------------------------------------------------------------
def load_data():
    events = pd.read_csv("athlete_events.csv")
    regions = pd.read_csv("noc_regions.csv")

    df = events.merge(regions, on="NOC", how="left")
    df.rename(columns={"region": "Country"}, inplace=True)

    # Fill missing country names with NOC code
    df["Country"] = df["Country"].fillna(df["NOC"])

    # Medal flag
    df["HasMedal"] = df["Medal"].notna().astype(int)

    # Fill numeric gaps with median per sport (reasonable for body metrics)
    for col in ["Age", "Height", "Weight"]:
        df[col] = df.groupby("Sport")[col].transform(lambda s: s.fillna(s.median()))

    print("Shape:", df.shape)
    print(df.info())
    return df


# ------------------------------------------------------------------
# 2. MEDAL TALLY BY COUNTRY
# ------------------------------------------------------------------
def medal_tally(df):
    # Count each team medal once per country/event/year (not once per athlete)
    medals = df.dropna(subset=["Medal"]).drop_duplicates(
        subset=["Team", "NOC", "Games", "Event", "Medal"]
    )
    tally = (
        medals.groupby(["Country", "Medal"]).size().unstack(fill_value=0)
        .reindex(columns=["Gold", "Silver", "Bronze"], fill_value=0)
    )
    tally["Total"] = tally.sum(axis=1)
    tally = tally.sort_values(["Gold", "Silver", "Bronze"], ascending=False)

    print("\nTop 15 countries by medals:\n", tally.head(15))
    tally.to_csv("outputs/medal_tally.csv")

    top = tally.head(15)[["Gold", "Silver", "Bronze"]]
    top.plot(kind="barh", stacked=True, figsize=(10, 7),
             color=["#FFD700", "#C0C0C0", "#CD7F32"])
    plt.gca().invert_yaxis()
    plt.title("Top 15 Countries - All-time Olympic Medals")
    plt.xlabel("Medals")
    plt.tight_layout()
    plt.savefig("outputs/top_countries_medals.png", dpi=150)
    plt.close()
    return medals, tally


# ------------------------------------------------------------------
# 3. PERFORMANCE OVER TIME
# ------------------------------------------------------------------
def trends_over_time(df, medals, countries=("USA", "Russia", "China", "Germany", "UK")):
    # Participation growth
    part = df.groupby(["Year", "Season"])["ID"].nunique().reset_index()
    plt.figure(figsize=(11, 5))
    sns.lineplot(data=part, x="Year", y="ID", hue="Season", marker="o")
    plt.title("Number of Athletes per Olympic Games")
    plt.ylabel("Unique athletes")
    plt.tight_layout()
    plt.savefig("outputs/participation_over_time.png", dpi=150)
    plt.close()

    # Medals per year for selected countries
    sub = medals[medals["Country"].isin(countries)]
    by_year = sub.groupby(["Year", "Country"]).size().reset_index(name="Medals")
    plt.figure(figsize=(11, 6))
    sns.lineplot(data=by_year, x="Year", y="Medals", hue="Country", marker="o")
    plt.title("Medals Won Over Time - Selected Countries")
    plt.tight_layout()
    plt.savefig("outputs/country_medal_trends.png", dpi=150)
    plt.close()


# ------------------------------------------------------------------
# 4. GENDER ANALYSIS
# ------------------------------------------------------------------
def gender_analysis(df):
    g = df.groupby(["Year", "Sex"])["ID"].nunique().unstack(fill_value=0)
    g["Female_%"] = g["F"] / (g["F"] + g["M"]) * 100

    plt.figure(figsize=(11, 5))
    plt.plot(g.index, g["Female_%"], marker="o", color="purple")
    plt.title("Female Participation (%) Over Time")
    plt.ylabel("% of athletes")
    plt.xlabel("Year")
    plt.tight_layout()
    plt.savefig("outputs/female_participation.png", dpi=150)
    plt.close()


# ------------------------------------------------------------------
# 5. SPORT-LEVEL ANALYSIS
# ------------------------------------------------------------------
def sport_analysis(df):
    # Which sports give the most medals?
    top_sports = (
        df[df["HasMedal"] == 1].groupby("Sport").size().sort_values(ascending=False).head(15)
    )
    plt.figure(figsize=(10, 6))
    sns.barplot(x=top_sports.values, y=top_sports.index, palette="viridis")
    plt.title("Top 15 Sports by Medals Awarded (athlete-level)")
    plt.xlabel("Medals")
    plt.tight_layout()
    plt.savefig("outputs/top_sports.png", dpi=150)
    plt.close()

    # Body profile of medalists in popular sports
    popular = ["Gymnastics", "Swimming", "Athletics", "Basketball", "Weightlifting", "Football"]
    sub = df[df["Sport"].isin(popular) & (df["HasMedal"] == 1)]
    plt.figure(figsize=(10, 6))
    sns.scatterplot(data=sub, x="Height", y="Weight", hue="Sport", alpha=0.5)
    plt.title("Height vs Weight of Medalists by Sport")
    plt.tight_layout()
    plt.savefig("outputs/height_weight_medalists.png", dpi=150)
    plt.close()

    # Age distribution of medalists
    plt.figure(figsize=(10, 5))
    sns.boxplot(data=sub, x="Sport", y="Age")
    plt.title("Age Distribution of Medalists")
    plt.tight_layout()
    plt.savefig("outputs/age_medalists.png", dpi=150)
    plt.close()


# ------------------------------------------------------------------
# 6. EFFICIENCY: MEDAL CONVERSION RATE
# ------------------------------------------------------------------
def conversion_rate(df, min_athletes=500):
    stats = df.groupby("Country").agg(
        Athletes=("ID", "nunique"),
        Entries=("ID", "count"),
        Medals=("HasMedal", "sum"),
    )
    stats["MedalRate_%"] = stats["Medals"] / stats["Entries"] * 100
    stats = stats[stats["Athletes"] >= min_athletes].sort_values("MedalRate_%", ascending=False)

    print("\nMost efficient countries (medals per entry):\n", stats.head(10))
    stats.to_csv("outputs/country_efficiency.csv")

    stats.head(15)["MedalRate_%"].plot(kind="bar", figsize=(11, 5), color="teal")
    plt.title(f"Medal Conversion Rate (countries with >= {min_athletes} athletes)")
    plt.ylabel("Medals per 100 entries")
    plt.tight_layout()
    plt.savefig("outputs/conversion_rate.png", dpi=150)
    plt.close()


# ------------------------------------------------------------------
# 7. HOST-COUNTRY ADVANTAGE (simple version)
# ------------------------------------------------------------------
def host_advantage(medals):
    host_map = {
        "Los Angeles": "USA", "Atlanta": "USA", "Salt Lake City": "USA",
        "Beijing": "China", "London": "UK", "Sydney": "Australia",
        "Athens": "Greece", "Barcelona": "Spain", "Seoul": "South Korea",
        "Moskva": "Russia", "Munich": "Germany", "Rio de Janeiro": "Brazil",
        "Montreal": "Canada", "Mexico City": "Mexico", "Tokyo": "Japan",
    }
    rows = []
    for city, country in host_map.items():
        games = medals[medals["City"] == city]
        if games.empty:
            continue
        for year, g in games.groupby("Year"):
            host_medals = (g["Country"] == country).sum()
            prev = medals[(medals["Country"] == country) & (medals["Year"] == year - 4)]
            rows.append({"City": city, "Year": year, "Host": country,
                         "Medals_as_host": host_medals, "Medals_prev_games": len(prev)})
    host_df = pd.DataFrame(rows)
    print("\nHost advantage:\n", host_df)
    host_df.to_csv("outputs/host_advantage.csv", index=False)


# ------------------------------------------------------------------
# MAIN
# ------------------------------------------------------------------
if __name__ == "__main__":
    data = load_data()
    medals_df, tally_df = medal_tally(data)
    trends_over_time(data, medals_df)
    gender_analysis(data)
    sport_analysis(data)
    conversion_rate(data)
    host_advantage(medals_df)
    print("\nDone! Charts and CSVs saved in the 'outputs' folder.")
